import React, { useState } from "react";
import { useSignIn } from "@clerk/expo";
import {
  SafeAreaView,
  ScrollView,
  View,
  Text,
  TextInput,
  Pressable,
  StyleSheet,
  KeyboardAvoidingView,
  Platform,
  StatusBar,
} from "react-native";
import { Link, router } from "expo-router";
import { Ionicons } from "@expo/vector-icons";

export default function LoginScreen() {
  const { signIn, fetchStatus } = useSignIn();
  const [isLoading, setIsLoading] = useState(false);
  const [email, setEmail] = useState("");
  const [password, setPassword] = useState("");
  const [showPassword, setShowPassword] = useState(false);
  const [error, setError] = useState("");

  async function handleLogin() {
    setError("");

    const cleanEmail = email.trim();

    if (!cleanEmail || !password) {
      setError("Please enter your email and password.");
      return;
    }

    if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(cleanEmail)) {
      setError("Please enter a valid email address.");
      return;
    }

    if (fetchStatus === "fetching") {
      return;
    }

    setIsLoading(true);

    try {
      const { error: createError } = await signIn.create({
        identifier: cleanEmail,
      });

      if (createError) {
        setError(createError.message || "Unable to start sign-in.");
        return;
      }

      const { error: passwordError } = await signIn.password({
        password,
      });

      if (passwordError) {
        setError(passwordError.message || "Unable to sign in.");
        return;
      }

      if (signIn.status === "complete") {
        const { error: finalizeError } = await signIn.finalize();

        if (finalizeError) {
          setError(finalizeError.message || "Could not complete sign-in.");
        }
      } else {
        setError(
          "Additional verification is required. Please contact support while we configure this step."
        );
      }
    } catch (error) {
      setError(error?.message || "Something went wrong. Please try again.");
    } finally {
      setIsLoading(false);
    }
  }

  return (
    <SafeAreaView style={styles.screen}>
      <StatusBar barStyle="light-content" backgroundColor={COLORS.navy} />
      <KeyboardAvoidingView
        style={styles.keyboard}
        behavior={Platform.OS === "ios" ? "padding" : undefined}
      >
        <ScrollView
          contentContainerStyle={styles.scrollContent}
          keyboardShouldPersistTaps="handled"
          showsVerticalScrollIndicator={false}
        >
          <View style={styles.topSection}>
            <Pressable
              style={styles.backButton}
              onPress={() => router.replace("/")}
              accessibilityRole="button"
              accessibilityLabel="Back to welcome screen"
            >
              <Ionicons name="chevron-back" size={20} color={COLORS.white} />
              <Text style={styles.backText}>Back</Text>
            </Pressable>

            <View style={styles.logoContainer}>
              <Ionicons
                name="binoculars-outline"
                size={37}
                color={COLORS.navy}
              />
            </View>

            <Text style={styles.brand}>LectureLens</Text>
            <Text style={styles.topCaption}>Your learning, all in focus.</Text>
          </View>

          <View style={styles.formSection}>
            <Text style={styles.heading}>Welcome back</Text>
            <Text style={styles.subtitle}>
              Sign in to continue your learning journey.
            </Text>

            <View style={styles.form}>
              <Text style={styles.label}>Email address</Text>
              <View style={styles.inputContainer}>
                <Ionicons
                  name="mail-outline"
                  size={20}
                  color={COLORS.muted}
                  style={styles.inputIcon}
                />
                <TextInput
                  style={styles.input}
                  placeholder="Enter your university email"
                  placeholderTextColor={COLORS.placeholder}
                  value={email}
                  onChangeText={setEmail}
                  keyboardType="email-address"
                  autoCapitalize="none"
                  autoCorrect={false}
                  autoComplete="email"
                  textContentType="emailAddress"
                  accessibilityLabel="Email address"
                  returnKeyType="next"
                />
              </View>

              <Text style={styles.label}>Password</Text>
              <View style={styles.inputContainer}>
                <Ionicons
                  name="lock-closed-outline"
                  size={20}
                  color={COLORS.muted}
                  style={styles.inputIcon}
                />
                <TextInput
                  style={styles.input}
                  placeholder="Enter your password"
                  placeholderTextColor={COLORS.placeholder}
                  value={password}
                  onChangeText={setPassword}
                  secureTextEntry={!showPassword}
                  autoCapitalize="none"
                  autoComplete="current-password"
                  textContentType="password"
                  accessibilityLabel="Password"
                  returnKeyType="done"
                  onSubmitEditing={handleLogin}
                />
                <Pressable
                  onPress={() => setShowPassword((previous) => !previous)}
                  style={styles.passwordToggle}
                  accessibilityRole="button"
                  accessibilityLabel={showPassword ? "Hide password" : "Show password"}
                  hitSlop={8}
                >
                  <Ionicons
                    name={showPassword ? "eye-off-outline" : "eye-outline"}
                    size={20}
                    color={COLORS.navy}
                  />
                </Pressable>
              </View>

              {error ? (
                <Text
                  style={styles.error}
                  accessibilityRole="alert"
                  accessibilityLiveRegion="polite"
                >
                  {error}
                </Text>
              ) : null}

              <Pressable
                style={[
                  styles.primaryButton,
                  (isLoading || fetchStatus === "fetching") && styles.buttonDisabled,
                ]}
                onPress={handleLogin}
                disabled={isLoading || fetchStatus === "fetching"}
                accessibilityRole="button"
              >
                <Text style={styles.primaryButtonText}>
                  {isLoading ? "Signing In..." : "Sign In"}
                </Text>
                {!isLoading && (
                  <Ionicons name="arrow-forward" size={20} color={COLORS.navy} />
                )}
              </Pressable>

              <View style={styles.registerRow}>
                <Text style={styles.registerPrompt}>Don't have an account?</Text>
                <Link href="/register" asChild>
                  <Pressable accessibilityRole="button">
                    <Text style={styles.registerLink}>Create account</Text>
                  </Pressable>
                </Link>
              </View>
            </View>
          </View>

          <Text style={styles.footer}>Supporting accessible learning at MUBAS.</Text>
        </ScrollView>
      </KeyboardAvoidingView>
    </SafeAreaView>
  );
}

import React, { useState } from "react";
import {
  View,
  Text,
  Modal,
  TouchableOpacity,
  StyleSheet,
} from "react-native";

import { Ionicons } from "@expo/vector-icons";
import { router } from "expo-router";
import { useAuth, useUser } from "@clerk/expo";

import styles from "../../styles/homeStyles";

export default function HomeHeader({ userName }) {
  const [menuVisible, setMenuVisible] = useState(false);

  const { signOut } = useAuth();
  const { isLoaded, user } = useUser();

  // Prefer the actual logged-in Clerk user's identity.
  const clerkUserName =
    user?.firstName?.trim() ||
    user?.fullName?.trim() ||
    user?.username?.trim() ||
    user?.primaryEmailAddress?.emailAddress?.split("@")[0] ||
    "";

  // Keep userName as a fallback for compatibility with existing screens.
  const displayName = clerkUserName || userName?.trim() || "";

  const initial = isLoaded
    ? displayName.charAt(0).toUpperCase() || "S"
    : "S";

  const closeMenu = () => {
    setMenuVisible(false);
  };

  const navigateTo = (destination) => {
    closeMenu();
    router.push(destination);
  };

  const handleSignOut = async () => {
    try {
      closeMenu();

      await signOut();

      router.replace("/login");
    } catch (error) {
      console.error("Sign out failed:", error);
    }
  };

  return (
    <>
      <View style={styles.header}>
        {/* Initial of the authenticated user */}
        <View style={styles.usernameContainer}>
          <Text style={styles.userName}>{initial}</Text>
        </View>

        {/* LectureLens icon */}
        <Ionicons
          name="binoculars-outline"
          size={28}
          color="#FFFFFF"
        />

        {/* Menu button */}
        <TouchableOpacity
          onPress={() => setMenuVisible(true)}
          accessibilityRole="button"
          accessibilityLabel="Open navigation menu"
          activeOpacity={0.7}
        >
          <Ionicons
            name="menu-outline"
            size={28}
            color="#FFFFFF"
          />
        </TouchableOpacity>
      </View>

      {/* Navigation menu */}
      <Modal
        visible={menuVisible}
        transparent
        animationType="fade"
        onRequestClose={closeMenu}
      >
        <View style={menuStyles.overlay}>
          <TouchableOpacity
            style={menuStyles.backdrop}
            activeOpacity={1}
            onPress={closeMenu}
          />

          <View style={menuStyles.menu}>
            <View style={menuStyles.menuHeader}>
              <Text style={menuStyles.menuTitle}>
                LectureLens
              </Text>

              <TouchableOpacity
                onPress={closeMenu}
                accessibilityLabel="Close menu"
              >
                <Ionicons
                  name="close"
                  size={24}
                  color="#12325B"
                />
              </TouchableOpacity>
            </View>

            <View style={menuStyles.divider} />

            <MenuItem
              icon="folder-open-outline"
              title="Manage Modules"
              onPress={() => navigateTo("/modules")}
            />

            <MenuItem
              icon="document-text-outline"
              title="My Notes"
              onPress={() => navigateTo("/notes")}
            />

            <MenuItem
              icon="book-outline"
              title="References"
              onPress={() => navigateTo("/references")}
            />

            <MenuItem
              icon="settings-outline"
              title="Settings"
              onPress={() => navigateTo("/settings")}
            />

            <View style={menuStyles.divider} />

            <MenuItem
              icon="log-out-outline"
              title="Sign Out"
              onPress={handleSignOut}
              danger
            />
          </View>
        </View>
      </Modal>
    </>
  );
}

==================== components/home/FloatingRecordButton.jsx ====================


import React, { useState } from "react";
import {
    ActivityIndicator,
    Alert,
    Modal,
    ScrollView,
    Text,
    TouchableOpacity,
    View,
} from "react-native";
import { Ionicons } from "@expo/vector-icons";
import {
    AudioModule,
    RecordingPresets,
    setAudioModeAsync,
    useAudioRecorder,
} from "expo-audio";

import { api } from "../../services/api";
import styles from "../../styles/homeStyles";

const NAVY = "#12325B";
const GOLD = "#F2B544";
const MUTED = "#667085";

function getItems(result, key) {
    if (Array.isArray(result)) return result;
    if (Array.isArray(result?.[key])) return result[key];
    return [];
}

export default function FloatingRecordButton() {
    const recorder = useAudioRecorder(RecordingPresets.HIGH_QUALITY);

    const [isRecording, setIsRecording] = useState(false);
    const [isProcessing, setIsProcessing] = useState(false);
    const [selectionVisible, setSelectionVisible] = useState(false);
    const [loadingLectures, setLoadingLectures] = useState(false);
    const [lectures, setLectures] = useState([]);
    const [selectedLecture, setSelectedLecture] = useState(null);
    const [selectionError, setSelectionError] = useState("");

    async function openLectureSelection() {
        setSelectionVisible(true);
        setLoadingLectures(true);
        setSelectionError("");
        setLectures([]);

        try {
            const moduleResult = await api.listModules();
            const modules = getItems(moduleResult, "modules");

            if (modules.length === 0) {
                setSelectionError(
                    "You have no modules yet. Create a module and add a lecture first."
                );
                return;
            }

            const groups = await Promise.all(
                modules.map(async (module) => {
                    const result = await api.listLectures(module.id);
                    const moduleLectures = getItems(result, "lectures");

                    return moduleLectures.map((lecture) => ({
                        ...lecture,
                        moduleName: module.name,
                    }));
                })
            );

            const allLectures = groups.flat();

            setLectures(allLectures);

            if (allLectures.length === 0) {
                setSelectionError(
                    "You have no saved lectures. Add a lecture before recording."
                );
            }
        } catch (error) {
            setSelectionError(
                error.message || "Could not load your lectures. Please try again."
            );
        } finally {
            setLoadingLectures(false);
        }
    }

    async function startRecording(lecture) {
        try {
            setIsProcessing(true);

            const permission =
                await AudioModule.requestRecordingPermissionsAsync();

            if (!permission.granted) {
                Alert.alert(
                    "Microphone permission required",
                    "Allow LectureLens to access your microphone in your device settings before recording."
                );
                return;
            }

            await setAudioModeAsync({
                allowsRecording: true,
                playsInSilentMode: true,
            });

            await recorder.prepareToRecordAsync();
            recorder.record();

            setSelectedLecture(lecture);
            setIsRecording(true);
        } catch (error) {
            console.error("Start recording error:", error);

            Alert.alert(
                "Recording failed",
                error?.message ||
                    "LectureLens could not start recording. Please try again."
            );
        } finally {
            setIsProcessing(false);
        }
    }

    async function stopRecording() {
        try {
            setIsProcessing(true);

            await recorder.stop();
            setIsRecording(false);

            await setAudioModeAsync({
                allowsRecording: false,
                playsInSilentMode: true,
            });

            const audioUri = recorder.uri;

            if (!audioUri) {
                Alert.alert(
                    "Recording stopped",
                    "The recording stopped, but no audio file location was returned."
                );
                return;
            }

            if (!selectedLecture?.id) {
                Alert.alert(
                    "Lecture not selected",
                    `The recording is on your device at:\n\n${audioUri}`
                );
                return;
            }

            try {
                const title =
                    `${selectedLecture.title || "Lecture recording"} - Recording`;

                const result = await api.uploadRecording(
                    selectedLecture.id,
                    audioUri,
                    title
                );

                Alert.alert(
                    "Upload successful",
                    `Your recording has been uploaded to:\n${selectedLecture.title || "Selected lecture"}`
                );

                console.log("LectureLens recording upload result:", result);
            } catch (uploadError) {
                console.error("Recording upload error:", uploadError);

                Alert.alert(
                    "Upload failed",
                    `Your recording is still available on your device:\n\n${audioUri}\n\nReason: ${
                        uploadError.message || "Please try again."
                    }`
                );
            }
        } catch (error) {
            console.error("Stop recording error:", error);

            Alert.alert(
                "Could not stop recording",
                error?.message ||
                    "Something went wrong while stopping the recording."
            );
        } finally {
            setIsRecording(false);
            setIsProcessing(false);
        }
    }

    async function handlePress() {
        if (isProcessing) return;

        if (isRecording) {
            await stopRecording();
            return;
        }

        await openLectureSelection();
    }

    async function handleLectureSelect(lecture) {
        setSelectionVisible(false);
        await startRecording(lecture);
    }

    return (
        <>
            <View style={{ alignItems: "center" }}>
                <TouchableOpacity
                    style={[
                        styles.floatingButton,
                        isRecording && { backgroundColor: "#C62828" },
                        isProcessing && { opacity: 0.6 },
                    ]}
                    onPress={handlePress}
                    disabled={isProcessing}
                    accessibilityRole="button"
                    accessibilityLabel={
                        isRecording
                            ? "Stop and upload lecture recording"
                            : "Choose a lecture and start recording"
                    }
                >
                    <Ionicons
                        name={isRecording ? "stop" : "mic"}
                        size={32}
                        color="#FFFFFF"
                    />
                </TouchableOpacity>

                {isRecording && (
                    <Text
                        style={{
                            marginTop: 6,
                            color: "#C62828",
                            fontWeight: "600",
                        }}
                    >
                        Recording: {selectedLecture?.title || "Lecture"}
                    </Text>
                )}

                {isProcessing && (
                    <Text style={{ marginTop: 4, fontSize: 12 }}>
                        Please wait...
                    </Text>
                )}
            </View>

            <Modal
                visible={selectionVisible}
                transparent
                animationType="slide"
                onRequestClose={() => setSelectionVisible(false)}
            >
                <View
                    style={{
                        flex: 1,
                        justifyContent: "flex-end",
                        backgroundColor: "rgba(0,0,0,0.45)",
                    }}
                >
                    <View
                        style={{
                            maxHeight: "75%",
                            backgroundColor: "#F5F7FA",
                            borderTopLeftRadius: 20,
                            borderTopRightRadius: 20,
                            padding: 20,
                        }}
                    >
                        <View
                            style={{
                                flexDirection: "row",
                                alignItems: "center",
                                justifyContent: "space-between",
                                marginBottom: 8,
                            }}
                        >
                            <Text
                                style={{
                                    color: NAVY,
                                    fontSize: 21,
                                    fontWeight: "800",
                                    flex: 1,
                                }}
                            >
                                Choose a lecture
                            </Text>

                            <TouchableOpacity
                                onPress={() => setSelectionVisible(false)}
                                accessibilityRole="button"
                                accessibilityLabel="Close lecture selection"
                            >
                                <Ionicons
                                    name="close"
                                    size={28}
                                    color={NAVY}
                                />
                            </TouchableOpacity>
                        </View>

                        <Text
                            style={{
                                color: MUTED,
                                marginBottom: 14,
                                lineHeight: 20,
                            }}
                        >
                            Select the lecture to which this recording should
                            be saved.
                        </Text>

                        {loadingLectures ? (
                            <View style={{ padding: 28, alignItems: "center" }}>
                                <ActivityIndicator size="large" color={NAVY} />
                                <Text style={{ marginTop: 10, color: MUTED }}>
                                    Loading your lectures...
                                </Text>
                            </View>
                        ) : selectionError ? (
                            <View style={{ paddingVertical: 16 }}>
                                <Text
                                    style={{
                                        color: selectionError.includes("Could not")
                                            ? "#B42318"
                                            : MUTED,
                                        lineHeight: 21,
                                    }}
                                >
                                    {selectionError}
                                </Text>

                                {selectionError.includes("Could not") && (
                                    <TouchableOpacity
                                        onPress={openLectureSelection}
                                        style={{
                                            backgroundColor: NAVY,
                                            padding: 12,
                                            borderRadius: 9,
                                            alignItems: "center",
                                            marginTop: 14,
                                        }}
                                    >
                                        <Text
                                            style={{
                                                color: "#FFFFFF",
                                                fontWeight: "700",
                                            }}
                                        >
                                            Try again
                                        </Text>
                                    </TouchableOpacity>
                                )}

                                {!selectionError.includes("Could not") && (
                                    <TouchableOpacity
                                        onPress={() => setSelectionVisible(false)}
                                        style={{
                                            borderColor: NAVY,
                                            borderWidth: 1,
                                            padding: 12,
                                            borderRadius: 9,
                                            alignItems: "center",
                                            marginTop: 14,
                                        }}
                                    >
                                        <Text
                                            style={{
                                                color: NAVY,
                                                fontWeight: "700",
                                            }}
                                        >
                                            Close
                                        </Text>
                                    </TouchableOpacity>
                                )}
                            </View>
                        ) : (
                            <ScrollView
                                keyboardShouldPersistTaps="handled"
                                showsVerticalScrollIndicator={false}
                            >
                                {lectures.map((lecture) => (
                                    <TouchableOpacity
                                        key={String(lecture.id)}
                                        onPress={() => handleLectureSelect(lecture)}
                                        disabled={isProcessing}
                                        style={{
                                            backgroundColor: "#FFFFFF",
                                            borderWidth: 1,
                                            borderColor: GOLD,
                                            borderRadius: 12,
                                            padding: 15,
                                            marginBottom: 10,
                                        }}
                                    >
                                        <Text
                                            style={{
                                                color: NAVY,
                                                fontSize: 15,
                                                fontWeight: "700",
                                            }}
                                        >
                                            {lecture.title || "Untitled lecture"}
                                        </Text>

                                        <Text
                                            style={{
                                                color: MUTED,
                                                fontSize: 13,
                                                marginTop: 5,
                                            }}
                                        >
                                            {lecture.moduleName}
                                        </Text>

                                        {!!lecture.topic && (
                                            <Text
                                                style={{
                                                    color: MUTED,
                                                    fontSize: 13,
                                                    marginTop: 4,
                                                }}
                                            >
                                                {lecture.topic}
                                            </Text>
                                        )}

                                        <Text
                                            style={{
                                                color: NAVY,
                                                fontWeight: "700",
                                                marginTop: 10,
                                            }}
                                        >
                                            Select and record
                                        </Text>
                                    </TouchableOpacity>
                                ))}
                            </ScrollView>
                        )}

                        <TouchableOpacity
                            onPress={() => setSelectionVisible(false)}
                            style={{
                                alignItems: "center",
                                padding: 13,
                                marginTop: 8,
                            }}
                        >
                            <Text style={{ color: NAVY, fontWeight: "600" }}>
                                Cancel
                            </Text>
                        </TouchableOpacity>
                    </View>
                </View>
            </Modal>
        </>
    );
}
